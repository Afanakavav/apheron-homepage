// Firestore Security Rules for APHERON
// Copy and paste these rules into Firebase Console > Firestore Database > Rules

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // Projects collection - only authenticated users can read/write
    match /projects/{projectId} {
      // Allow read for authenticated users only
      allow read: if request.auth != null;
      
      // Allow write (create, update, delete) for authenticated users only
      allow create, update, delete: if request.auth != null;
    }
    
    // Deny all other collections by default
    match /{document=**} {
      allow read, write: if false;
    }
  }
}

/* 
EXPLANATION:
1. Only authenticated users (logged in via Firebase Auth) can access projects
2. Public users (not logged in) cannot see or modify anything
3. All other collections are completely blocked

TO APPLY THESE RULES:
1. Go to Firebase Console
2. Click "Firestore Database" in the left menu
3. Click the "Rules" tab
4. Replace the existing rules with the rules above
5. Click "Publish"

OPTIONAL: For more security, you can restrict to specific user emails:
  allow read, write: if request.auth.token.email == "francesco.perone00@gmail.com";
*/
